### 0.1 의존성 그래프 생성 (30분)

**목표**: 154개 파일의 의존 관계 파악 → 최적 머징 순서 결정

```bash
# 1. Both-modified 파일 목록 생성 (이미 있다면 스킵)
git diff --name-only merge/cline-v3.32.7-to-v3.34.0 HEAD | \
  grep -E "^(src|webview-ui)/" > /tmp/both-modified.txt

# 2. Import 관계 분석 스크립트 생성
cat > /tmp/analyze-imports.js << 'EOF'
const fs = require('fs');
const path = require('path');

const files = fs.readFileSync('/tmp/both-modified.txt', 'utf-8')
  .split('\n')
  .filter(f => f.trim());

const graph = {};

files.forEach(file => {
  if (!fs.existsSync(file)) return;

  const content = fs.readFileSync(file, 'utf-8');
  const imports = [];

  // import 문 추출
  const importRegex = /import\s+.*\s+from\s+['"]([^'"]+)['"]/g;
  let match;
  while ((match = importRegex.exec(content)) !== null) {
    const importPath = match[1];

    // 상대 경로 정규화
    if (importPath.startsWith('.')) {
      const dir = path.dirname(file);
      const resolved = path.resolve(dir, importPath);
      const normalized = resolved.replace(process.cwd() + '/', '') +
                        (fs.existsSync(resolved + '.ts') ? '.ts' :
                         fs.existsSync(resolved + '.tsx') ? '.tsx' : '');

      // both-modified 파일만
      if (files.includes(normalized)) {
        imports.push(normalized);
      }
    }
  }

  graph[file] = {
    imports_from: imports,
    imported_by: []
  };
});

// imported_by 역계산
Object.keys(graph).forEach(file => {
  graph[file].imports_from.forEach(dep => {
    if (graph[dep]) {
      graph[dep].imported_by.push(file);
    }
  });
});

// 의존성 레벨 계산
function calculateLevel(file, visited = new Set()) {
  if (visited.has(file)) return 999; // 순환 의존성
  visited.add(file);

  const imports = graph[file]?.imports_from || [];
  if (imports.length === 0) return 0;

  return 1 + Math.max(...imports.map(dep => calculateLevel(dep, new Set(visited))));
}

Object.keys(graph).forEach(file => {
  graph[file].level = calculateLevel(file);
});

fs.writeFileSync('.work-logs/dependency-graph.json', JSON.stringify(graph, null, 2));
console.log('✅ Dependency graph generated: .work-logs/dependency-graph.json');

// 레벨별 요약
const levels = {};
Object.keys(graph).forEach(file => {
  const level = graph[file].level;
  levels[level] = (levels[level] || 0) + 1;
});
console.log('📊 Levels:', levels);
EOF

node /tmp/analyze-imports.js
```

**목표**: 각 파일의 충돌 가능성 예측 → P0-P3 우선순위 부여

```bash
# 1. CARET MODIFICATION 패턴 분석
grep -r "CARET MODIFICATION" src/ webview-ui/ --include="*.ts" --include="*.tsx" -n > \
  .work-logs/caret-modifications.txt

# 2. Cline v3.34.0 실제 변경 확인 (Cline 저장소 필요)
# 만약 Cline 저장소가 없다면 GitHub에서 diff 확인

# 3. 충돌 위험도 매트릭스 생성
cat > /tmp/assess-risk.js << 'EOF'
const fs = require('fs');
const graph = JSON.parse(fs.readFileSync('.work-logs/dependency-graph.json'));
const caretMods = fs.readFileSync('.work-logs/caret-modifications.txt', 'utf-8');

const matrix = [];

Object.keys(graph).forEach(file => {
  const caretModCount = (caretMods.match(new RegExp(file, 'g')) || []).length;

  // 위험도 산정
  let risk = 0;
  if (caretModCount === 0) risk = 5; // 0% - Caret 수정 없음
  else if (caretModCount <= 2) risk = 20; // 20% - 소량 수정
  else if (caretModCount <= 5) risk = 40; // 40% - 중간 수정
  else if (caretModCount <= 10) risk = 60; // 60% - 많은 수정
  else if (caretModCount <= 20) risk = 80; // 80% - 대규모 수정
  else risk = 95; // 95% - 매우 많은 수정

  // Critical 파일 식별
  const critical = [
    'src/extension.ts',
    'src/core/task/index.ts',
    'src/core/controller/index.ts',
    'src/core/context/external-rules/external-rules.ts',
    'src/integrations/cli-subagents/subagent_command.ts'
  ];

  if (critical.includes(file)) risk = Math.max(risk, 95);

  // 우선순위
  let priority = 'P3';
  if (risk >= 95) priority = 'P0';
  else if (risk >= 60) priority = 'P1';
  else if (risk >= 20) priority = 'P2';

  matrix.push({
    file,
    caret_mods: caretModCount,
    risk,
    priority,
    level: graph[file].level
  });
});

// 정렬: priority (P0 먼저), then level (낮은 레벨 먼저)
matrix.sort((a, b) => {
  if (a.priority !== b.priority) {
    return a.priority.localeCompare(b.priority);
  }
  return a.level - b.level;
});

fs.writeFileSync('.work-logs/conflict-risk-matrix.json', JSON.stringify(matrix, null, 2));

// 마크다운 테이블 생성
let md = '# 충돌 위험도 매트릭스\n\n';
md += '| 파일 | Caret 수정 | 위험도 | 우선순위 | 레벨 |\n';
md += '|------|-----------|--------|---------|------|\n';
matrix.forEach(m => {
  md += `| ${m.file} | ${m.caret_mods}회 | ${m.risk}% | ${m.priority} | ${m.level} |\n`;
});

md += '\n## 통계\n\n';
const stats = {
  P0: matrix.filter(m => m.priority === 'P0').length,
  P1: matrix.filter(m => m.priority === 'P1').length,
  P2: matrix.filter(m => m.priority === 'P2').length,
  P3: matrix.filter(m => m.priority === 'P3').length
};
md += `- P0 (Critical): ${stats.P0}개 (${(stats.P0/matrix.length*100).toFixed(1)}%)\n`;
md += `- P1 (High): ${stats.P1}개 (${(stats.P1/matrix.length*100).toFixed(1)}%)\n`;
md += `- P2 (Medium): ${stats.P2}개 (${(stats.P2/matrix.length*100).toFixed(1)}%)\n`;
md += `- P3 (Low): ${stats.P3}개 (${(stats.P3/matrix.length*100).toFixed(1)}%)\n`;

fs.writeFileSync('.work-logs/conflict-risk-matrix.md', md);
console.log('✅ Risk matrix generated');
console.log('📊 Stats:', stats);
EOF

node /tmp/assess-risk.js
```

### 0.4 자동화 스크립트 준비 (20분)

```bash

cat > scripts/3way-diff.sh << 'EOF'
#!/bin/bash
# 3-way diff 보기
# Usage: ./scripts/3way-diff.sh src/extension.ts

FILE=$1

echo "========================================"
echo "🔍 3-WAY DIFF: $FILE"
echo "========================================"

echo ""
echo "=== 1️⃣ 충돌 소스 (현재 파일, conflict markers 있음) ==="

echo ""
echo ""
echo "=== 2️⃣ 캐럿 소스 (caret-main) ==="

echo ""
echo ""
echo "=== 3️⃣ 클라인 소스 (cline-latest) ==="
git show cline-latest:"$FILE" 2>/dev/null || echo "❌ File not found in cline-latest"

echo ""
echo "========================================"
echo "💡 분석 방법:"
echo "  1. 캐럿이 뭘 추가했는가? (2번 소스 확인)"
echo "  2. 클라인이 뭘 변경했는가? (3번 소스 확인)"
echo "  3. 충돌 지점에서 어떻게 통합할까? (1번 소스 수정)"

chmod +x scripts/3way-diff.sh

# 2. 스테이지 머징 스크립트
cat > scripts/stage-merge.sh << 'EOF'
#!/bin/bash
# 특정 스테이지 머징
# Usage: ./scripts/stage-merge.sh 1

STAGE=$1
FILES=$(jq -r ".[$STAGE].files[]" .work-logs/merge-stages.json)

echo "🔄 Merging Stage $STAGE..."

for file in $FILES; do
  echo "📄 Processing: $file"

  # cline-latest 브랜치에서 파일 머징
  git merge-file "$file" \
    <(git show caret-main:$file 2>/dev/null) \
    <(git show cline-latest:$file 2>/dev/null) \
    2>&1 | tee -a .work-logs/merge-log-stage-$STAGE.txt

  if [ $? -eq 0 ]; then
    echo "  ✅ Auto-merge success"
  else
    echo "  ⚠️ Conflict detected - needs manual resolution"
    echo "  💡 Run: ./scripts/3way-diff.sh $file"
  fi
done

echo ""
echo "✅ Stage $STAGE merge complete"
echo "📝 Log: .work-logs/merge-log-stage-$STAGE.txt"
EOF

chmod +x scripts/stage-merge.sh
```

### 🚨 문제 해결 프로토콜

**문제 발생 시 순서**:

1. **절대 하지 말 것**:
   - ❌ 컴파일 에러를 코드 수정으로 해결하려 하지 마라
   - ❌ 타입 에러를 타입 정의 추가로 우회하지 마라
   - ❌ "왜 안 되지?"라고 생각하며 코드를 읽지 마라

2. **반드시 할 것**:
   - ✅ 문제 파일 식별
   - ✅ 3-way diff 보기: `./scripts/3way-diff.sh <file>`
   - ✅ 각 변경의 **의도** 이해:
     ```
     캐럿 소스 (2번): Caret이 추가한 기능은?
     클라인 소스 (3번): Cline이 개선한 점은?
     ```
   - ✅ 통합 전략 결정:
     ```
     Option A: Cline 구조 + Caret 기능 재적용
     Option B: Caret 구조 + Cline 개선사항 병합
     Option C: 독립적 변경이면 둘 다 유지
     ```
   - ✅ 수동 편집: `code <file>`

3. **의사결정 가이드**:
   ```
   질문 1: Cline의 변경이 구조적 개선인가? (리팩토링, 아키텍처 변경)
     → Yes: Cline 구조 채택 (Option A)
     → No: 다음 질문으로

   질문 2: Caret의 변경이 독립적인가? (import 추가, 새 함수 추가)
     → Yes: 둘 다 유지 (Option C)
     → No: 다음 질문으로

   질문 3: 두 변경이 같은 위치를 수정하는가?
     → Yes: 수동 통합 필요 (Option A 또는 B)
     → No: 둘 다 유지 (Option C)
   ```
