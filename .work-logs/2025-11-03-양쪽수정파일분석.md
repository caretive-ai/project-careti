```typescript
// CARET MODIFICATION: F11 - Input History with backend persistence
export function usePersistentInputHistory() {
  const [history, setHistory] = useState<string[]>([])

  // Load from backend on mount
  useEffect(() => {
    CaretGlobalManager.getInputHistory().then(setHistory)
  }, [])

  // Save to backend on change
  const addToHistory = useCallback((input: string) => {
    const newHistory = [input, ...history].slice(0, 50)
    setHistory(newHistory)
    CaretGlobalManager.setInputHistory(newHistory)
  }, [history])

  return { history, addToHistory }
}
```
